cmake_minimum_required(VERSION 3.0.2)
project(humanoid_navigation)
find_package(catkin REQUIRED COMPONENTS roscpp rospy plan_env cv_bridge message_filters std_msgs geometry_msgs nav_msgs sensor_msgs visualization_msgs message_generation)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
catkin_python_setup()
add_message_files(FILES HumanoidCommand.msg HeightState.msg RawVoxelMap.msg)
generate_messages(DEPENDENCIES std_msgs)
catkin_package(CATKIN_DEPENDS message_runtime std_msgs geometry_msgs nav_msgs sensor_msgs visualization_msgs roscpp rospy plan_env)
include_directories(${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
add_executable(humanoid_static_grid_map src/static_grid_map.cpp)
target_compile_features(humanoid_static_grid_map PRIVATE cxx_std_14)
target_link_libraries(humanoid_static_grid_map ${catkin_LIBRARIES})
add_dependencies(humanoid_static_grid_map ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
install(TARGETS humanoid_static_grid_map RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
catkin_install_python(PROGRAMS scripts/navigation_node.py scripts/executor_node.py tests/ros_acceptance.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  find_package(rostest REQUIRED)
  find_package(PythonInterp 3 REQUIRED)
  add_test(NAME humanoid_core COMMAND ${PYTHON_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_core.py)
  foreach(scene open narrow low compound blocked feedback_loss replan_failure)
    add_rostest(tests/${scene}.test)
  endforeach()
endif()
